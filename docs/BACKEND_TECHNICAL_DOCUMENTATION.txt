SECTION 1: PROJECT OVERVIEW
==================================================

Platform Type   : SEO-optimized blog platform with admin panel
Backend Role    : REST API server, business logic, auth, image pipeline, analytics
Deployment      : VPS / Railway / Render
Database        : PostgreSQL (via Sequelize ORM + sequelize-typescript)

Image Storage   : GitHub Contents API + jsDelivr CDN

SECTION 5: MODEL DEFINITIONS
==================================================

--------------------------------------------------
Model: User
--------------------------------------------------
Table: users

Fields:
  id              : UUID, PK, defaultValue UUIDV4
  name            : STRING(100), allowNull false
  email           : STRING(255), allowNull false, unique true
  password_hash   : STRING(255), allowNull false
  role            : ENUM('superadmin','editor'), defaultValue 'editor'
  avatar_url      : STRING(500), allowNull true
  is_active       : BOOLEAN, defaultValue true
  last_login_at   : DATE (TIMESTAMPTZ), allowNull true
  created_at      : DATE (TIMESTAMPTZ), allowNull false, defaultValue NOW
  updated_at      : DATE (TIMESTAMPTZ), allowNull false, defaultValue NOW

Instance Methods:
  validatePassword(plain: string): Promise<boolean>
    → bcrypt.compare(plain, this.password_hash)

Hooks:
  beforeCreate: hash password with bcrypt, saltRounds=12
  beforeUpdate: if password_hash changed, re-hash

Associations:
  hasMany RefreshToken (foreignKey: user_id, onDelete: 'CASCADE')
  hasMany Blog (foreignKey: author_id, onDelete: 'RESTRICT')
  hasMany Image (foreignKey: uploaded_by, onDelete: 'RESTRICT')


--------------------------------------------------
Model: RefreshToken
--------------------------------------------------
Table: refresh_tokens

Fields:
  id              : UUID, PK, defaultValue UUIDV4
  user_id         : UUID, FK → users, allowNull false (ON DELETE CASCADE)
  token_hash      : STRING(255), allowNull false, unique true
  expires_at      : DATE (TIMESTAMPTZ), allowNull false
  ip_address      : STRING(45), allowNull true
  user_agent      : TEXT, allowNull true
  created_at      : DATE (TIMESTAMPTZ), allowNull false, defaultValue NOW
  updated_at      : DATE (TIMESTAMPTZ), allowNull false, defaultValue NOW

Associations:
  belongsTo User (foreignKey: user_id, onDelete: 'CASCADE')


--------------------------------------------------
Model: Category
--------------------------------------------------
Table: categories

Fields:
  id              : UUID, PK, defaultValue UUIDV4
  name            : STRING(100), allowNull false, unique true
  slug            : STRING(120), allowNull false, unique true
  description     : TEXT, allowNull true
  meta_title      : STRING(160), allowNull true
  meta_description: STRING(320), allowNull true
  is_active       : BOOLEAN, defaultValue true
  sort_order      : INTEGER, defaultValue 0
  created_at      : DATE (TIMESTAMPTZ), allowNull false, defaultValue NOW
  updated_at      : DATE (TIMESTAMPTZ), allowNull false, defaultValue NOW

Hooks:
  beforeValidate: auto-generate slug from name if not provided

Associations:
  hasMany Blog (foreignKey: category_id, onDelete: 'SET_NULL')


--------------------------------------------------
Model: Tag
--------------------------------------------------
Table: tags

Fields:
  id              : UUID, PK, defaultValue UUIDV4
  name            : STRING(100), allowNull false, unique true
  slug            : STRING(120), allowNull false, unique true
  created_at      : DATE (TIMESTAMPTZ), allowNull false, defaultValue NOW
  updated_at      : DATE (TIMESTAMPTZ), allowNull false, defaultValue NOW

Hooks:
  beforeValidate: auto-generate slug from name

Associations:
  belongsToMany Blog through BlogTag (onDelete: 'CASCADE')


--------------------------------------------------
Model: Blog
--------------------------------------------------
Table: blogs

Fields:
  id              : UUID, PK, defaultValue UUIDV4
  title           : STRING(300), allowNull false
  slug            : STRING(320), allowNull false, unique true
  excerpt         : TEXT, allowNull true
  content_html    : TEXT, allowNull false
  author_id       : UUID, FK → users, allowNull false (ON DELETE RESTRICT)
  category_id     : UUID, FK → categories, allowNull true (ON DELETE SET NULL)
  status          : ENUM('draft','published'), defaultValue 'draft'
  is_featured     : BOOLEAN, defaultValue false
  published_at    : DATE (TIMESTAMPTZ), allowNull true
  read_time_minutes: INTEGER, defaultValue 0
  seo_title       : STRING(160), allowNull true
  seo_description : STRING(320), allowNull true
  canonical_url   : STRING(500), allowNull true
  focus_keyword   : STRING(100), allowNull true
  structured_data : JSONB, allowNull true
  og_image_url    : STRING(500), allowNull true
  og_title        : STRING(160), allowNull true
  og_description  : STRING(320), allowNull true
  created_at      : DATE (TIMESTAMPTZ), allowNull false, defaultValue NOW
  updated_at      : DATE (TIMESTAMPTZ), allowNull false, defaultValue NOW

Hooks:
  beforeCreate: slugify title if slug not provided
  beforeCreate: calculate read_time_minutes from content_html
  beforeUpdate: recalculate read_time_minutes if content_html changed
  afterCreate: create BlogAnalytics record for this blog

Associations:
  belongsTo User (as: author, foreignKey: author_id, onDelete: 'RESTRICT')
  belongsTo Category (foreignKey: category_id, onDelete: 'SET_NULL')
  belongsToMany Tag through BlogTag (onDelete: 'CASCADE')
  hasOne BlogAnalytics (onDelete: 'CASCADE')
  hasMany Image (foreignKey: blog_id, onDelete: 'SET_NULL')
  hasMany PageViewLog (foreignKey: blog_id, onDelete: 'CASCADE')


--------------------------------------------------
Model: BlogTag
--------------------------------------------------
Table: blog_tags

Fields:
  blog_id         : UUID, FK → blogs, allowNull false (ON DELETE CASCADE)
  tag_id          : UUID, FK → tags, allowNull false (ON DELETE CASCADE)

Primary Key: [blog_id, tag_id]

  height          : INTEGER, allowNull false

  mime_type       : STRING(50), defaultValue 'image/webp'
  alt_text        : STRING(300), allowNull true

  created_at      : DATE (TIMESTAMPTZ), allowNull false, defaultValue NOW
  updated_at      : DATE (TIMESTAMPTZ), allowNull false, defaultValue NOW

Associations:
  belongsTo Blog (foreignKey: blog_id, onDelete: 'SET_NULL')

SECTION 6: API ROUTES SPECIFICATION
==================================================

Base URL: /api/v1

--------------------------------------------------
AUTH ROUTES: /api/v1/auth
--------------------------------------------------

POST   /login
  Body        : { email, password }
  Response    : { accessToken, user: { id, name, role } }
  Action      : Validate credentials, sign JWT access token,
                generate refresh token, store hashed token in DB,
                set refreshToken in HTTP-only cookie

POST   /logout
  Auth        : Authenticated (access token)
  Response    : { message }
  Action      : Delete refresh token from DB, clear cookie

POST   /refresh
  Cookie      : refreshToken
  Response    : { accessToken }
  Action      : Validate refresh token from cookie,
                compare hash, issue new access token

GET    /me
  Auth        : Authenticated
  Response    : { id, name, email, role, avatar_url }


--------------------------------------------------
USER ROUTES: /api/v1/users
--------------------------------------------------
All routes: Auth required + superadmin role

GET    /
  Response    : Paginated list of users

POST   /
  Body        : { name, email, password, role }
  Action      : Create new editor/admin account

PATCH  /:id
  Body        : { name, avatar_url, is_active }
  Action      : Update user profile

PATCH  /:id/password
  Body        : { current_password, new_password }
  Action      : Change password with bcrypt validation

DELETE /:id
  Action      : Soft deactivate (is_active = false)

--------------------------------------------------
BLOG ROUTES: /api/v1/blogs
--------------------------------------------------

Admin CRUD:

GET    /                        (Auth: editor+)
  Query       : ?status=draft|published&page=1&limit=20&search=
  Response    : Paginated blogs with author, category, tag counts

GET    /:id                     (Auth: editor+)
  Response    : Full blog with tags, images, analytics, author

POST   /                        (Auth: editor+)
  Body        : {
                  title, slug?, excerpt, content_html,

                  status, is_featured, published_at?,
                  seo_title, seo_description, canonical_url,

  Action      : Sanitize content_html, auto-slug if not provided,
                calculate read_time_minutes, upsert tags,
                create blog_tags associations

PATCH  /:id                     (Auth: editor+)
  Body        : Partial blog fields (same as POST)
  Action      : Update blog, sync tags, recalculate read_time

DELETE /:id                     (Auth: superadmin)
  Action      : Delete blog, cascade images metadata (NOT GitHub files)

PATCH  /:id/status              (Auth: editor+)
  Body        : { status: 'draft'|'published', published_at? }

SECTION 7: AUTHENTICATION ARCHITECTURE
==================================================

Access Token:
  Type        : JWT (signed with JWT_ACCESS_SECRET)
  Payload     : { sub: userId, role, iat, exp }
  Expiry      : 15 minutes
  Transport   : Authorization: Bearer <token> header

Refresh Token:
  Type        : Opaque random string (crypto.randomBytes(64).toString('hex'))
  Storage     : SHA-256 hash stored in refresh_tokens table
  Expiry      : 7 days
  Transport   : HTTP-only, Secure, SameSite=Strict cookie

Token Flow (Login):
  1. Validate email + password
  2. Generate accessToken (JWT, 15m)
  3. Generate raw refreshToken (random 64 bytes)
  4. Hash refreshToken with SHA-256
  5. Store hash + expires_at + ip + user_agent in refresh_tokens
  6. Update user.last_login_at
  7. Return accessToken in body
  8. Set refreshToken in HTTP-only cookie

Token Flow (Refresh):
  1. Read refreshToken cookie
  2. Hash it with SHA-256
  3. Find matching row in refresh_tokens
  4. Validate expiry
  5. Verify user.is_active
  6. Issue new accessToken
  7. Optionally rotate refresh token (recommended)

Token Flow (Logout):
  1. Read refreshToken cookie
  2. Hash it
  3. Delete from refresh_tokens
  4. Clear cookie

Middleware: authenticate.ts
  1. Extract Bearer token from Authorization header
  2. Verify with jwt.verify(token, JWT_ACCESS_SECRET)
  3. Attach decoded payload to req.user
  4. Reject with 401 if missing or invalid

Middleware: authorize.ts
  Usage       : authorize('superadmin') or authorize('editor', 'superadmin')
  Action      : Check req.user.role against allowed roles
  Reject      : 403 Forbidden

SECTION 8: IMAGE HANDLING PIPELINE
==================================================

Endpoint: POST /api/v1/images/upload
Middleware: multer (memoryStorage, no disk writes)

Step-by-step pipeline in image.service.ts:

1. RECEIVE
   - multer receives multipart/form-data into buffer (memory)
   - Field: file (required), blog_id (optional), alt_text (optional)

2. AUTHENTICATE
   - authenticate middleware runs before multer
   - req.user must be present

3. VALIDATE MIME TYPE
   - Check file.mimetype against ALLOWED_MIME_TYPES env array
   - Reject with 415 if not allowed

4. VALIDATE FILE SIZE
   - Check file.size against MAX_IMAGE_SIZE_BYTES
   - Reject with 413 if exceeded

5. GENERATE METADATA
   - slugify blog title or fallback to timestamp
   - storagePath = posts/{YYYY}/{MM}/{blog-slug}/
   - Generate a unique timestamp-based filename (e.g. {timestamp}-{random}.webp)

6. PROCESS IMAGE WITH SHARP
   - Limit max width to 1600px for desktop/mobile responsive optimization (maintaining aspect ratio)
   - .rotate() → auto-orient from EXIF
   - .withMetadata(false) → strip all EXIF metadata
   - Convert to webp with quality 82
   - .toBuffer() → get processed buffer
   - .metadata() → capture output width, height, and file size

7. UPLOAD TO GITHUB
   github.service.ts:
   - filePath = storagePath + filename
   - content  = buffer.toString('base64')
   - PUT https://api.github.com/repos/{owner}/{repo}/contents/{filePath}
     Headers: Authorization: Bearer GITHUB_TOKEN
     Body   : { message: 'upload: {filePath}', content, branch }
   - On conflict (422): GET existing SHA, then PUT with sha field to update

8. GENERATE CDN URL
   cdn.service.ts:
   - cdnUrl = JSDELIVR_BASE + '/' + filePath
   Example:
   https://cdn.jsdelivr.net/gh/owner/repo@main/posts/2026/05/blog-slug/1716382900-xyz.webp

9. SAVE TO DATABASE
   Insert Image record:
   - blog_id, github_path, cdn_url,
     width, height, file_size_bytes, mime_type, alt_text, uploaded_by

10. RETURN RESPONSE
    {
      success: true,
      data: {
        id,
        cdn_url,
        width,
        height,
        alt_text
      }
    }

Error handling at each step:
  - Multer error → 400
  - MIME/size error → 413/415
  - Sharp error → 422 with message
  - GitHub API error → 502 Bad Gateway, log full error
  - DB error → 500, optionally clean up GitHub files

SECTION 12: ERROR HANDLING ARCHITECTURE
==================================================

Custom error classes in types/errors.ts:

  class AppError extends Error:
    statusCode   : number
    errorCode    : string
    isOperational: boolean
    errors       : unknown

    Constructor: (statusCode, errorCode, message, errors = null, isOperational = true)

  class ValidationError extends AppError (400, 'VALIDATION_ERROR')
  class UnauthorizedError extends AppError (401, 'UNAUTHORIZED')
  class ForbiddenError extends AppError (403, 'FORBIDDEN')
  class NotFoundError extends AppError (404, 'NOT_FOUND')
  class ConflictError extends AppError (409, 'CONFLICT')
  class UpstreamError extends AppError (502, 'UPSTREAM_ERROR') -- GitHub API failures

Global error handler middleware (errorHandler.ts):
  - If AppError: respond with statusCode + message
  - If Sequelize.ValidationError: extract messages → 400
  - If Sequelize.UniqueConstraintError: → 409
  - If JsonWebTokenError: → 401
  - If MulterError: → 400 or 413
  - Unknown errors: log full stack → respond 500
  - Persist to DB (ErrorLog table):
    - All crucial server errors (500), database errors, rate limit blocks, and third-party failures.
  - Critical Alerting:
    - If severity is HIGH, CRITICAL, or FATAL, asynchronously trigger email notification to the administrator via nodemailer (email.service.ts).

Standard API response wrapper (utils/apiResponse.ts):

  Class: ApiResponse

  static success(res: Response, statusCode: number, message: string, data: unknown = null, meta: unknown = null): Response
    → Sends standardized success payload:
      {
        success: true,
        status_code: statusCode,
        message: string,
        data: any,
        meta: any,
        timestamp: string (ISO)
      }

  static error(res: Response, statusCode: number, message: string, errors: unknown = null, fileName: string = 'N/A', functionName: string = 'N/A', correlationId: string = 'N/A'): Response
    - Logs errors (statusCode >= 400) to logger.error, including correlation ID, file, and function names.
    → Sends standardized error payload:
      {
        success: false,
        status_code: statusCode,
        message: string,
        correlation_id: string,
        errors: any,
        debug_info?: { file: string, function: string }, (only in non-production environments)
        timestamp: string (ISO)
      }

  static paginated(res: Response, statusCode: number, message: string, data: unknown[], pagination: { total: number, page: number, limit: number, total_pages: number }): Response
    → Sends standardized paginated payload:
      {
        success: true,
        status_code: statusCode,
        message: string,
        data: Array,
        pagination: {
          total: number,
          page: number,
          limit: number,
          total_pages: number
        },
        timestamp: string (ISO)
      }

  static handleControllerError(res: Response, req: Request, error: unknown, fileName: string, functionName: string): Response
    - Centralized controller catch block utility.
    - Resolves statusCode/message/details from AppError or defaults to 500 (Internal Server Error).
    - Dispatches to ApiResponse.error using Request correlation ID.
