==================================================
DATABASE DESIGN
==================================================
PostgreSQL + Sequelize ORM (sequelize-typescript)
==================================================


==================================================
TABLE: users
==================================================
Purpose: Admin/author accounts
--------------------------------------------------
id                  UUID            PRIMARY KEY DEFAULT gen_random_uuid()
name                VARCHAR(100)    NOT NULL
email               VARCHAR(255)    NOT NULL UNIQUE
password_hash       VARCHAR(255)    NOT NULL
role                ENUM            NOT NULL DEFAULT 'editor' -- 'superadmin', 'editor'
avatar_url          VARCHAR(500)    NULL
is_active           BOOLEAN         NOT NULL DEFAULT true
last_login_at       TIMESTAMP       NULL
created_at          TIMESTAMP       NOT NULL DEFAULT NOW()
updated_at          TIMESTAMP       NOT NULL DEFAULT NOW()


==================================================
TABLE: refresh_tokens
==================================================
Purpose: JWT refresh token management
--------------------------------------------------
id                  UUID            PRIMARY KEY DEFAULT gen_random_uuid()
user_id             UUID            NOT NULL REFERENCES users(id) ON DELETE CASCADE
token_hash          VARCHAR(255)    NOT NULL UNIQUE
expires_at          TIMESTAMP       NOT NULL
ip_address          VARCHAR(45)     NULL
user_agent          TEXT            NULL
created_at          TIMESTAMP       NOT NULL DEFAULT NOW()

INDEX: user_id
INDEX: token_hash


==================================================
TABLE: categories
==================================================
Purpose: Blog categories
--------------------------------------------------
id                  UUID            PRIMARY KEY DEFAULT gen_random_uuid()
name                VARCHAR(100)    NOT NULL UNIQUE
slug                VARCHAR(120)    NOT NULL UNIQUE
description         TEXT            NULL
meta_title          VARCHAR(160)    NULL
meta_description    VARCHAR(320)    NULL
is_active           BOOLEAN         NOT NULL DEFAULT true
sort_order          INTEGER         NOT NULL DEFAULT 0
created_at          TIMESTAMP       NOT NULL DEFAULT NOW()
updated_at          TIMESTAMP       NOT NULL DEFAULT NOW()

INDEX: slug


==================================================
TABLE: tags
==================================================
Purpose: Blog tags
--------------------------------------------------
id                  UUID            PRIMARY KEY DEFAULT gen_random_uuid()
name                VARCHAR(100)    NOT NULL UNIQUE
slug                VARCHAR(120)    NOT NULL UNIQUE
created_at          TIMESTAMP       NOT NULL DEFAULT NOW()

INDEX: slug


==================================================
TABLE: blogs
==================================================
Purpose: Core blog post table
--------------------------------------------------
id                  UUID            PRIMARY KEY DEFAULT gen_random_uuid()
title               VARCHAR(300)    NOT NULL
slug                VARCHAR(320)    NOT NULL UNIQUE
excerpt             TEXT            NULL
content_html        TEXT            NOT NULL
author_id           UUID            NOT NULL REFERENCES users(id) ON DELETE RESTRICT
category_id         UUID            NULL REFERENCES categories(id) ON DELETE SET NULL
status              ENUM            NOT NULL DEFAULT 'draft'  -- 'draft', 'published'
is_featured         BOOLEAN         NOT NULL DEFAULT false
published_at        TIMESTAMP       NULL
read_time_minutes   INTEGER         NOT NULL DEFAULT 0
-- SEO Fields
seo_title           VARCHAR(160)    NULL
seo_description     VARCHAR(320)    NULL
canonical_url       VARCHAR(500)    NULL
focus_keyword       VARCHAR(100)    NULL
structured_data     JSONB           NULL      -- JSON-LD schema markup
-- OG / Social
og_image_url        VARCHAR(500)    NULL
og_title            VARCHAR(160)    NULL
og_description      VARCHAR(320)    NULL
-- Timestamps
created_at          TIMESTAMP       NOT NULL DEFAULT NOW()
updated_at          TIMESTAMP       NOT NULL DEFAULT NOW()

INDEX: slug
INDEX: status
INDEX: category_id
INDEX: author_id
INDEX: published_at
INDEX: is_featured


==================================================
TABLE: blog_tags
==================================================
Purpose: Many-to-many join between blogs and tags
--------------------------------------------------
blog_id             UUID            NOT NULL REFERENCES blogs(id) ON DELETE CASCADE
tag_id              UUID            NOT NULL REFERENCES tags(id) ON DELETE CASCADE

PRIMARY KEY (blog_id, tag_id)

INDEX: tag_id


==================================================
TABLE: images
==================================================
Purpose: All uploaded images with CDN metadata
--------------------------------------------------
id                  UUID            PRIMARY KEY DEFAULT gen_random_uuid()
blog_id             UUID            NULL REFERENCES blogs(id) ON DELETE SET NULL
github_path         VARCHAR(500)    NOT NULL   -- posts/2026/05/blog-slug/image.webp
cdn_url             VARCHAR(500)    NOT NULL
width               INTEGER         NOT NULL
height              INTEGER         NOT NULL
file_size_bytes     INTEGER         NOT NULL
mime_type           VARCHAR(50)     NOT NULL DEFAULT 'image/webp'
alt_text            VARCHAR(300)    NULL
uploaded_by         UUID            NOT NULL REFERENCES users(id) ON DELETE RESTRICT
created_at          TIMESTAMP       NOT NULL DEFAULT NOW()

INDEX: blog_id


==================================================
TABLE: blog_analytics
==================================================
Purpose: Per-blog stats tracking
--------------------------------------------------
id                  UUID            PRIMARY KEY DEFAULT gen_random_uuid()
blog_id             UUID            NOT NULL REFERENCES blogs(id) ON DELETE CASCADE UNIQUE
total_views         BIGINT          NOT NULL DEFAULT 0
unique_visitors     BIGINT          NOT NULL DEFAULT 0
avg_read_time_sec   INTEGER         NOT NULL DEFAULT 0
bounce_count        BIGINT          NOT NULL DEFAULT 0
last_viewed_at      TIMESTAMP       NULL
updated_at          TIMESTAMP       NOT NULL DEFAULT NOW()

INDEX: blog_id


==================================================
TABLE: page_view_logs
==================================================
Purpose: Raw hit logs for analytics aggregation
--------------------------------------------------
id                  BIGSERIAL       PRIMARY KEY
blog_id             UUID            NOT NULL REFERENCES blogs(id) ON DELETE CASCADE
visitor_hash        VARCHAR(64)     NOT NULL   -- hashed IP+UA fingerprint
referrer            VARCHAR(500)    NULL
read_time_sec       INTEGER         NOT NULL DEFAULT 0
is_bounce           BOOLEAN         NOT NULL DEFAULT false
viewed_at           TIMESTAMP       NOT NULL DEFAULT NOW()

INDEX: blog_id
INDEX: viewed_at
INDEX: visitor_hash


==================================================
TABLE: search_logs
==================================================
Purpose: Track search queries for analytics
--------------------------------------------------
id                  BIGSERIAL       PRIMARY KEY
query               VARCHAR(300)    NOT NULL
result_count        INTEGER         NOT NULL DEFAULT 0
searched_at         TIMESTAMP       NOT NULL DEFAULT NOW()

INDEX: query
INDEX: searched_at


==================================================
TABLE: ad_units
==================================================
Purpose: Ad scripts and their placement config
--------------------------------------------------
id                  UUID            PRIMARY KEY DEFAULT gen_random_uuid()
name                VARCHAR(150)    NOT NULL
placement           ENUM            NOT NULL
                    -- 'HOME_TOP', 'HOME_MIDDLE', 'HOME_BOTTOM',
                    -- 'SEARCH_TOP', 'SEARCH_INLINE', 'SEARCH_BOTTOM',
                    -- 'BLOG_TOP', 'BLOG_INLINE_1', 'BLOG_INLINE_2',
                    -- 'BLOG_BOTTOM', 'BLOG_SIDEBAR'
ad_script           TEXT            NOT NULL   -- raw HTML/JS ad code
device_target       ENUM            NOT NULL DEFAULT 'all'  -- 'all', 'desktop', 'mobile'
is_active           BOOLEAN         NOT NULL DEFAULT true
sort_order          INTEGER         NOT NULL DEFAULT 0
created_at          TIMESTAMP       NOT NULL DEFAULT NOW()
updated_at          TIMESTAMP       NOT NULL DEFAULT NOW()

INDEX: placement
INDEX: is_active


==================================================
TABLE: error_logs
==================================================
Purpose: Database logging for crucial system and runtime errors
--------------------------------------------------
id                  UUID            PRIMARY KEY DEFAULT gen_random_uuid()
correlation_id      VARCHAR(50)     NULL
error_code          VARCHAR(50)     NULL
error_message       TEXT            NOT NULL
stack_trace         TEXT            NULL
file_name           VARCHAR(255)    NULL
function_name       VARCHAR(255)    NULL
severity            ENUM            NOT NULL DEFAULT 'MEDIUM' -- 'LOW', 'MEDIUM', 'HIGH', 'CRITICAL', 'FATAL'
error_type          ENUM            NOT NULL -- 'SERVER_ERROR', 'DATABASE_ERROR', 'RATE_LIMIT_ERROR', 'THIRD_PARTY_ERROR', 'OTHER'
request_url         VARCHAR(500)    NULL
request_method      VARCHAR(10)     NULL
ip_address          VARCHAR(45)     NULL
user_id             UUID            NULL REFERENCES users(id) ON DELETE SET NULL
created_at          TIMESTAMP       NOT NULL DEFAULT NOW()
updated_at          TIMESTAMP       NOT NULL DEFAULT NOW()

INDEX: error_type
INDEX: severity
INDEX: correlation_id
INDEX: created_at


==================================================
RELATIONSHIPS SUMMARY
==================================================

users ──< blogs                   (one author → many blogs)
users ──< refresh_tokens          (one user → many tokens)
users ──< images                  (one user → many uploads)
users ──< error_logs              (one user → many error logs)
categories ──< blogs              (one category → many blogs)
blogs ──< blog_tags >── tags      (many-to-many)
blogs ──── blog_analytics         (one-to-one)
blogs ──< images                  (one blog → many uploaded images)
blogs ──< page_view_logs          (one blog → many view hits)


==================================================
ENUM DEFINITIONS SUMMARY
==================================================

user_role:          superadmin | editor
blog_status:        draft | published
device_target:      all | desktop | mobile
error_severity:     LOW | MEDIUM | HIGH | CRITICAL | FATAL
error_type:         SERVER_ERROR | DATABASE_ERROR | RATE_LIMIT_ERROR | THIRD_PARTY_ERROR | OTHER
ad_placement:       HOME_TOP | HOME_MIDDLE | HOME_BOTTOM
                    SEARCH_TOP | SEARCH_INLINE | SEARCH_BOTTOM
                    BLOG_TOP | BLOG_INLINE_1 | BLOG_INLINE_2
                    BLOG_BOTTOM | BLOG_SIDEBAR


==================================================
NOTES
==================================================

- UUIDs used across all primary keys for security and scalability
- blog_analytics is updated via batch aggregation from page_view_logs
- page_view_logs can be periodically archived/pruned after aggregation
- structured_data (JSONB) stores JSON-LD for rich search results
- visitor_hash is a hashed fingerprint; no raw PII stored
- cdn_url is the jsDelivr CDN URL generated after GitHub upload
- github_path stores the raw storage path for deletion/management
- All slug fields are indexed for fast URL-based lookups
- read_time_minutes on blogs is computed on save from content_html length
